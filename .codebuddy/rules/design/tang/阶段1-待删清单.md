```ts
export class StoryManager {
    constructor() {}

    // 旧主线 stage 系统已废弃。本类降级为通用文字提示工具。
    update() {
        // 屏震和红屏的衰减已搬到 state.fx，并由迷宫/竞技场各自驱动。
        // 这里保留空 update，避免 game.ts 调用链断裂。
    }

        state.alertMsg = msg;
        state.alertColor = color;

            state.alertMsg = '';
        }, duration) as any;
    }
}
```
